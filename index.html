async function exportToImage() { const wrapper = document.getElementById('capture-wrapper'); wrapper.classList.add('is-exporting'); try { const currentOuterBg = getComputedStyle(document.documentElement).getPropertyValue('--outer-bg-color').trim(); const canvas = await html2canvas(wrapper, { scale: 2, useCORS: true, backgroundColor: currentOuterBg || '#f4f4f9' }); const dataUrl = canvas.toDataURL('image/png'); // 💡 [추가] 우측 상단 텍스트(playlist-title) 값 읽기 및 파일명 결정 로직 const rawTitle = document.getElementById('playlist-title').innerText.trim(); // 파일명으로 사용할 수 없는 특수문자(\ / : * ? " < > |) 제거 const safeTitle = rawTitle.replace(/[\\/:*?"<>|]/g, '').trim(); let fileName = 'theme playlist.png'; // 기본 파일명 // 기본값이 아니거나 사용자가 텍스트를 입력했을 경우 해당 텍스트로 저장명 설정 if (safeTitle && safeTitle !== 'Character Theme Tracks') { fileName = `${safeTitle}.png`; } if (isMobileOrIOS()) { const saveModal = document.getElementById('saveModal'); const saveModalImg = document.getElementById('saveModalImg'); saveModalImg.src = dataUrl; saveModal.classList.add('active'); } else { const link = document.createElement('a'); link.download = fileName; // 💡 동적으로 설정된 파일명 적용 link.href = dataUrl; link.click(); } } catch (err) { console.error('이미지 저장 중 오류가 발생했습니다:', err); alert('이미지 저장에 실패했습니다.'); } finally { wrapper.classList.remove('is-exporting'); } }
